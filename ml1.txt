import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.ensemble import RandomForestRegressor

from sklearn.metrics import (
    r2_score,
    mean_squared_error,
    mean_absolute_error
)

# Load dataset
df = pd.read_csv(
    r"C:\Users\purva\OneDrive\Desktop\lp3\dataset\uber.csv (1)\uber.csv"
)

print("Dataset Shape:", df.shape)

print("Dataset shape:", df.shape)
display(df.head())
df.info()

print("\nMissing Values:")
print(df.isnull().sum())






# Remove unnecessary columns
df = df.drop(columns=["Unnamed: 0", "key"], errors="ignore")

# Convert pickup_datetime into datetime format
df["pickup_datetime"] = pd.to_datetime(
    df["pickup_datetime"], errors="coerce"
)

# Remove missing values
df = df.dropna()

# Remove invalid records
df = df[
    (df["fare_amount"] > 0) &
    (df["passenger_count"] > 0) &
    (df["pickup_longitude"].between(-75, -72)) &
    (df["dropoff_longitude"].between(-75, -72)) &
    (df["pickup_latitude"].between(40, 42)) &
    (df["dropoff_latitude"].between(40, 42))
].copy()

# Extract time features
df["hour"] = df["pickup_datetime"].dt.hour
df["day"] = df["pickup_datetime"].dt.day
df["month"] = df["pickup_datetime"].dt.month

print("Dataset after preprocessing:", df.shape)
print("\nRemaining Missing Values:")
print(df.isnull().sum())







# Calculate Q1 and Q3
Q1 = df["fare_amount"].quantile(0.25)
Q3 = df["fare_amount"].quantile(0.75)

# Calculate IQR
IQR = Q3 - Q1

# Calculate lower and upper limits
lower = Q1 - 1.5 * IQR
upper = Q3 + 1.5 * IQR

# Identify outliers
outliers = df[
    (df["fare_amount"] < lower) |
    (df["fare_amount"] > upper)
]

print("Number of Outliers:", len(outliers))

# Remove outliers
df = df[
    (df["fare_amount"] >= lower) &
    (df["fare_amount"] <= upper)
].copy()

print("Dataset after removing outliers:", df.shape)

# Visualize outliers using boxplot
plt.figure(figsize=(7, 4))
sns.boxplot(x=df["fare_amount"])
plt.title("Fare Amount - Outlier Detection")
plt.show()







# Select numerical columns
numeric_df = df.select_dtypes(include="number")

# Calculate correlation
correlation = numeric_df.corr()

print("Correlation Matrix:")
print(correlation)

# Visualize correlation
plt.figure(figsize=(10, 7))

sns.heatmap(
    correlation,
    annot=True,
    cmap="coolwarm",
    fmt=".2f"
)

plt.title("Correlation Heatmap")
plt.show()







# Select input features
X = df[
    [
        "pickup_longitude",
        "pickup_latitude",
        "dropoff_longitude",
        "dropoff_latitude",
        "passenger_count",
        "hour",
        "day",
        "month"
    ]
]

# Select target variable
y = df["fare_amount"]

print("Features and target selected successfully!")
print("X shape:", X.shape)
print("y shape:", y.shape)








print("Starting model training...")

# Use a smaller sample to speed up training
X_sample = X.sample(n=min(20000, len(X)), random_state=42)
y_sample = y.loc[X_sample.index]

# Split dataset
X_train, X_test, y_train, y_test = train_test_split(
    X_sample, y_sample, test_size=0.2, random_state=42
)

# Linear Regression
print("\nTraining Linear Regression...")
lr = LinearRegression()
lr.fit(X_train, y_train)
lr_pred = lr.predict(X_test)

print("Linear Regression completed!")

# Random Forest
print("\nTraining Random Forest...")
rf = RandomForestRegressor(
    n_estimators=20,
    random_state=42,
    n_jobs=-1
)
rf.fit(X_train, y_train)
rf_pred = rf.predict(X_test)

print("Random Forest completed!")

# Evaluation
lr_r2 = r2_score(y_test, lr_pred)
lr_rmse = np.sqrt(mean_squared_error(y_test, lr_pred))
lr_mae = mean_absolute_error(y_test, lr_pred)

rf_r2 = r2_score(y_test, rf_pred)
rf_rmse = np.sqrt(mean_squared_error(y_test, rf_pred))
rf_mae = mean_absolute_error(y_test, rf_pred)

# Comparison
comparison = pd.DataFrame({
    "Model": ["Linear Regression", "Random Forest"],
    "R2 Score": [lr_r2, rf_r2],
    "RMSE": [lr_rmse, rf_rmse],
    "MAE": [lr_mae, rf_mae]
})

print("\nFINAL MODEL COMPARISON:")
print(comparison.round(3))

if rf_r2 > lr_r2:
    print("\nRandom Forest performs better based on R2 Score.")
else:
    print("\nLinear Regression performs better based on R2 Score.")
